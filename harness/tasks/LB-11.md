```ts
// Extracts the JSON object text from a model reply: drops <think> blocks and markdown fences.
export function extractJsonText(raw: string): string {
  let text = raw.replace(/<think>[\s\S]*?<\/think>/gi, '').trim();
  const fenced = /^```(?:json)?\s*([\s\S]*?)\s*```$/i.exec(text);
  if (fenced?.[1] !== undefined) text = fenced[1].trim();
  if (!text.startsWith('{')) {
    const start = text.indexOf('{');
    const end = text.lastIndexOf('}');
    if (start >= 0 && end > start) text = text.slice(start, end + 1);
  }
  return text;
}
```

```ts
// MODEL_EXTRA_BODY: optional JSON object merged into every request body, e.g. {"response_format":{"type":"json_object"}}.
export function parseExtraBody(raw: string): Record<string, unknown> {
  if (!raw) return {};
  let value: unknown;
  try { value = JSON.parse(raw); } catch { throw new Error('MODEL_EXTRA_BODY must be a JSON object'); }
  if (typeof value !== 'object' || value === null || Array.isArray(value)) throw new Error('MODEL_EXTRA_BODY must be a JSON object');
  const extra = {...(value as Record<string, unknown>)};
  delete extra.model; delete extra.messages; delete extra.stream;
  return extra;
}
```

```ts
// Usage: npx tsx scripts/live-smoke.ts R1   (the API must already be running)
import { RunSchema } from '@fairtrip/contracts';
const fixtureId = process.argv[2] ?? 'R1';
const base = process.env.SMOKE_BASE_URL ?? 'http://127.0.0.1:3001';
const started = await fetch(`${base}/api/runs`, {method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({fixtureId})});
if (!started.ok) { console.error(`POST /api/runs failed: HTTP ${started.status}`); process.exit(1); }
const {id} = await started.json() as {id: string};
const seen = new Set<number>();
for (let attempt = 0; attempt < 100; attempt++) {
  await new Promise(resolve => setTimeout(resolve, 1000));
  const run = RunSchema.parse(await (await fetch(`${base}/api/runs/${id}?after=0`)).json());
  for (const event of run.events) if (!seen.has(event.sequence)) { seen.add(event.sequence); console.log(`#${event.sequence} ${event.actor} ${event.type}: ${event.summary}`); }
  if (run.status === 'completed') { console.log(`RESULT completed: ${run.result.remedyId}, ${run.result.action.amountCents} cents, ruling ${run.result.ruling}, confidence ${run.result.confidence}`); process.exit(0); }
  if (run.status === 'incomplete') { console.log(`RESULT incomplete: ${run.missingEvidence.join(' | ')}`); process.exit(0); }
  if (run.status === 'failed') { console.log(`RESULT failed: ${run.error.code} ${run.error.message}`); process.exit(1); }
}
console.log('RESULT timeout after 100 s');
process.exit(1);
```

````ts
import { describe, expect, it } from 'vitest';
import { extractJsonText } from '@fairtrip/agents';
import { createModelAdapter, parseExtraBody } from '../../apps/api/src/model.js';
import { readConfig } from '../../apps/api/src/config.js';

describe('model output hardening', () => {
  it('extracts JSON from fences, think blocks and surrounding prose', () => {
    expect(extractJsonText('```json\n{"a":1}\n```')).toBe('{"a":1}');
    expect(extractJsonText('<think>x</think>{"a":1}')).toBe('{"a":1}');
    expect(extractJsonText('Here: {"a":1} thanks')).toBe('{"a":1}');
    expect(extractJsonText('not json')).toBe('not json');
  });
  it('parses MODEL_EXTRA_BODY safely', () => {
    expect(parseExtraBody('')).toEqual({});
    expect(parseExtraBody('{"response_format":{"type":"json_object"},"model":"x"}')).toEqual({response_format: {type: 'json_object'}});
    expect(() => parseExtraBody('[1]')).toThrow('MODEL_EXTRA_BODY');
  });
  it('merges extra body options without overriding model, messages or stream', async () => {
    const bodies: unknown[] = [];
    const fetcher: typeof fetch = async (_url, init) => {
      bodies.push(JSON.parse(String(init?.body)));
      return new Response(JSON.stringify({choices: [{message: {content: '{"ok":true}'}}]}), {status: 200});
    };
    const config = readConfig({RUN_MODE: 'live', MODEL_API_KEY: 'k', MODEL_BASE_URL: 'https://example.com/v1', MODEL_NAME: 'm', MODEL_EXTRA_BODY: '{"response_format":{"type":"json_object"},"model":"evil"}'});
    await createModelAdapter(config, fetcher).complete([{role: 'user', content: 'Return JSON'}], new AbortController().signal);
    expect(bodies[0]).toMatchObject({model: 'm', stream: false, response_format: {type: 'json_object'}});
  });
````

## Troubleshooting

- Every run fails right after the two `model.start` events with `Model request failed (HTTP 400)`, while `npm run check:model` passes → MiniMax rejects the advocates' system-only first request (error 2013). Do task LB-11A, which fixes it and then finishes A10.
- `output.repair` events or `produced unparseable output` → human checks `MODEL_EXTRA_BODY` contains `"reasoning_split":true`, restarts the server, retry.
- HTTP 401 → wrong key, or key and endpoint from different MiniMax regions; 404 → `MODEL_NAME` is not exactly `MiniMax-M3`, or wrong base URL; 400 → an option in `MODEL_EXTRA_BODY` is not accepted (go back to `{"reasoning_split":true}`); 429 → the shared Token Plan quota window is used up, wait. In all these cases STOP and report the exact line.
- `Run deadline exceeded (90s)` or per-call aborts → MiniMax-M3 is thinking too long. Human adds `"thinking":{"type":"disabled"}` to `MODEL_EXTRA_BODY`, restarts, retry; STOP if it still times out.
