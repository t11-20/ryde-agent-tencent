2. `src/hooks/useRunPolling.ts` — copy exactly (tested logic):
```ts
import { useCallback, useEffect, useRef, useState } from 'react';
import type { ActivityEvent, Run } from '@fairtrip/contracts';
import { ContractMismatchError, fetchRun, startRun } from '../lib/api.js';
import { maxSequence, mergeEvents } from '../lib/events.js';

export type PollPhase = 'idle' | 'starting' | 'polling' | 'done' | 'stopped' | 'start_failed' | 'contract_mismatch' | 'connection_lost';
export interface PollState { phase: PollPhase; run: Run | null; events: ActivityEvent[]; problem: string | null }
const POLL_MS = 1000;
const MAX_NETWORK_ERRORS = 5;
const TERMINAL: ReadonlySet<string> = new Set(['completed', 'incomplete', 'failed']);

export function useRunPolling() {
  const [state, setState] = useState<PollState>({phase: 'idle', run: null, events: [], problem: null});
  const generation = useRef(0);
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const runId = useRef<string | null>(null);
  const events = useRef<ActivityEvent[]>([]);
  const networkErrors = useRef(0);
  const clearTimer = () => { if (timer.current !== null) { clearTimeout(timer.current); timer.current = null; } };

  const pollOnce = useCallback(async (gen: number): Promise<void> => {
    const id = runId.current;
    if (id === null || gen !== generation.current) return;
    try {
      const run = await fetchRun(id, maxSequence(events.current));
      if (gen !== generation.current) return;
      networkErrors.current = 0;
      events.current = mergeEvents(events.current, run.events);
      const done = TERMINAL.has(run.status);
      setState({phase: done ? 'done' : 'polling', run, events: events.current, problem: null});
      if (!done) timer.current = setTimeout(() => { void pollOnce(gen); }, POLL_MS);
    } catch (error) {
      if (gen !== generation.current) return;
      const message = error instanceof Error ? error.message : String(error);
      if (error instanceof ContractMismatchError) { setState(s => ({...s, phase: 'contract_mismatch', problem: message})); return; }
      networkErrors.current += 1;
      if (networkErrors.current >= MAX_NETWORK_ERRORS) { setState(s => ({...s, phase: 'connection_lost', problem: message})); return; }
      timer.current = setTimeout(() => { void pollOnce(gen); }, POLL_MS);
    }
  }, []);

  const start = useCallback(async (fixtureId: string, riderClaim?: string): Promise<void> => {
    generation.current += 1;
    const gen = generation.current;
    clearTimer();
    runId.current = null; events.current = []; networkErrors.current = 0;
    setState({phase: 'starting', run: null, events: [], problem: null});
    try {
      const {id} = await startRun(riderClaim === undefined ? {fixtureId} : {fixtureId, riderClaim});
      if (gen !== generation.current) return;
      runId.current = id;
      setState(s => ({...s, phase: 'polling'}));
      void pollOnce(gen);
    } catch (error) {
      if (gen !== generation.current) return;
      setState({phase: 'start_failed', run: null, events: [], problem: error instanceof Error ? error.message : String(error)});
    }
  }, [pollOnce]);

  const keepWaiting = useCallback(() => {
    networkErrors.current = 0;
    setState(s => ({...s, phase: 'polling', problem: null}));
    void pollOnce(generation.current);
  }, [pollOnce]);
  const stop = useCallback(() => { generation.current += 1; clearTimer(); setState(s => ({...s, phase: 'stopped'})); }, []);
  useEffect(() => () => { generation.current += 1; clearTimer(); }, []);
  return {state, start, keepWaiting, stop};
}
```
3. `src/hooks/useFixtures.ts`: on mount call `fetchFixtures()` once; return `{ data: FixturesResponse | null; error: string | null; loading: boolean }`; ignore the result if the component unmounted.
4. `src/styles.css`: start with the tokens from SPEC §12.3 as CSS variables on `:root`, then a CSS grid: `grid-template-columns: 360px 1fr 380px` for the three top columns and a full-width evidence row below. Minimum body font 16 px; panels white on the paper background with a 1 px `--rule` border; actor accents as a 4 px left border in the actor colour; `.chip` (monospace, rounded) and `.chip-unresolved` (dashed `--warn` border); a visible `:focus-visible` outline.
5. Components (props in brackets; use `buildViewModel` output, never raw API objects, except `DisputePanel`):
- `Header` (mode: string) — product name, "Ryde dispute resolution prototype", badges DEMONSTRATION POLICY, SYNTHETIC DATA, and `LIVE MODEL` or `STUB` from the fixtures response mode.
- `DisputePanel` (fixtures, selectedId, onSelect, claim, onClaimChange, busy, onResolve) — select showing `"{id}: {label}"` with category; editable rider claim (or "Claim appears when the run starts" when the fixture has none); read-only driver statement; **Resolve dispute** button disabled while `busy`.
- `StatusPanel` (phase, run status, problem, elapsed seconds, onRunAgain, onKeepWaiting, onStop, onExport) — every state from SPEC §12.2 with its own text. Elapsed time ticks once per second while running and shows "of 90 s".
- `AdvocatePanel` (title, actor colour, side: SidePanel, knownIds) — tool requests, retrieved IDs as chips, case summary, arguments and counterevidence each with chips, requested remedy (`remedyLabel`), missing facts. Before the case arrives show "Gathering evidence…".
- `JudgePanel` (result or null, knownIds, status) — ruling (`rulingLabel`), findings with chips, "Model-assessed confidence 86%" with a bar, reasoning, then "To the rider" and "To the driver" blocks.
- `ActionCard` (result) — render only when `vm.showAction`; remedy label, `formatSgd(amountCents)`, recipient, recommendation and the note "Amount calculated by the server from the evidence, not by the model."
- `Timeline` (events) — vertical line with one dot per event coloured by actor; each item shows actor label, `eventLabel`, `+elapsed s`, summary and chips. The `judge.started` item is drawn larger with the text "Both cases handed to the Judge".
- `EvidencePanel` (evidence by family, knownIds) — four columns GPS, Chat, Payment, History; each card has `id="ev-{ID}"`, ID, summary, "retrieved by rider / driver", chat text in quotation marks, tag badges, the instruction-like warning badge, and a `<details>` element with the facts as formatted JSON in a `<pre>`.
- `PolicyPanel` (policies) — each clause with `id="pol-{ID}"`, ID, rule text and remedy criteria, under the heading "Policy clauses retrieved (DEMONSTRATION POLICY)".
- `Chip` (id, kind: 'evidence' | 'policy', known: boolean) — a `<button>`; when known it scrolls to `ev-{id}` or `pol-{id}` with `scrollIntoView({behavior: 'smooth', block: 'center'})` and adds a 1.5 s highlight class; when unknown it uses `.chip-unresolved` with `title="Not among the retrieved records"`.
Never use `dangerouslySetInnerHTML`. React escapes text by default; keep it that way.
6. `App.tsx`: `useFixtures()` + `useRunPolling()`; keep the selected fixture and an edited-claim flag; on Resolve call `start(fixtureId, claimEdited ? claim : undefined)`; `const vm = state.run ? buildViewModel(state.run, state.events) : null`; render the layout of SPEC §12.1. Export uses `buildExport` and `downloadJson`.
7. `main.tsx`: import `./styles.css` and render `<App />` inside `<StrictMode>`.
8. Add a "Demo operator script" section to `apps/web/README.md`: start the API (`npm run dev`), start the UI (`npm run dev:web`), open `http://127.0.0.1:5173`, run R1, then N2, then X1, and what to point at in each.
